/**
 * Concurrency-Safe Webhook Receiver for Google Sheets CRM
 */
function doPost(e) {
  // Acquire lock to prevent race condition overwrites
  const lock = LockService.getScriptLock();
  const success = lock.tryLock(30000); // 30s timeout

  if (!success) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: "Server busy: lock timeout. Request should be retried."
    })).setMimeType(ContentService.MimeType.JSON);
  }

  try {
    const payload = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Leads");

    // Atomic read-write cycle
    const lastRow = sheet.getLastRow();
    const agent = assignNextAgent();

    sheet.getRange(lastRow + 1, 1, 1, 5).setValues([[
      new Date(),
      payload.name,
      payload.company,
      agent,
      "New"
    ]]);

    return ContentService.createTextOutput(JSON.stringify({ status: "success", leadId: lastRow + 1 }));
  } catch (err) {
    AuditLogger.logError("doPost_Exception", err);
    return ContentService.createTextOutput(JSON.stringify({ status: "fatal", error: err.toString() }));
  } finally {
    lock.releaseLock();
  }
}
/**
 * High-Speed Batch Range Operations
 * Eliminates 6-minute Google Apps Script runtime quotas
 */
function synchronizeCRMReports() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const source = ss.getSheetByName("Master_Leads");
  const target = ss.getSheetByName("Executive_Dashboard");

  // Read entire table in 1 single API call
  const data = source.getDataRange().getValues();
  const headers = data[0];
  const rows = data.slice(1);

  // In-memory aggregation
  const summary = {};
  rows.forEach(r => {
    const agent = r[3];
    summary[agent] = (summary[agent] || 0) + 1;
  });

  // Prepare 2D matrix for single batch write
  const outputMatrix = Object.entries(summary).map(([agent, count]) => [agent, count]);

  // Single batch write
  if (outputMatrix.length > 0) {
    target.getRange(2, 1, outputMatrix.length, 2).setValues(outputMatrix);
  }
}
/**
 * Round-Robin Agent Router with Load Balancing
 */
function assignNextAgent() {
  const props = PropertiesService.getScriptProperties();
  const agents = ["Agent Sarah", "Agent David", "Agent Maya"];
  let currentIndex = parseInt(props.getProperty("LAST_AGENT_INDEX") || "0", 10);

  const nextIndex = (currentIndex + 1) % agents.length;
  props.setProperty("LAST_AGENT_INDEX", nextIndex.toString());

  return agents[nextIndex];
}
